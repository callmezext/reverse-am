Dokumentasi REST API Alight Motion Premium Activator

← Web Activator
BASE URL https://ampro.enchant.id/api
GET /status

Cek ketersediaan dan status server activator.

Response (200 OK)
{
  "status": "online",
  "timestamp": "2026-09-11T00:24:18.676Z"
}
Contoh cURL
curl -s https://ampro.enchant.id/api/status
POST /send-link

Meminta Firebase Auth mengirimkan magic link login ke email target.

Headers
Content-Type: application/json
Request Body (JSON)
FieldTipeWajib?Keterangan
email string wajib Email valid yang akan didaftarkan/diaktifkan
Request Body Contoh
{
  "email": "user@gmail.com"
}
Response (200 OK)
{
  "success": true,
  "email": "user@gmail.com",
  "message": "link dikirim ke user@gmail.com. cek inbox / spam."
}
Contoh cURL
curl -X POST https://ampro.enchant.id/api/send-link \
  -H "Content-Type: application/json" \
  -d '{"email":"user@gmail.com"}'
POST /verify-link

Memvalidasi link email, mengatur custom nickname, dan mengaktifkan lisensi Annual VIP Pro.

Headers
Content-Type: application/json
Request Body (JSON)
FieldTipeWajib?Keterangan
email string wajib Email tujuan yang menerima magic link
magicLink string wajib URL lengkap dari email atau kode oobCode
nickname string opsional Custom nama akun. Jika kosong, otomatis diambil dari email
Request Body Contoh
{
  "email": "user@gmail.com",
  "magicLink": "https://alightcreative.com?ui_sid=...&oobCode=ab12cd...",
  "nickname": "Zenno Pro"
}
Response (200 OK)
{
  "success": true,
  "message": "verifikasi berhasil, premium aktif.",
  "data": {
    "displayName": "Zenno Pro",
    "email": "user@gmail.com",
    "uid": "1B2c3D4e5F6g...",
    "status": "ACTIVE",
    "membershipStatus": "PREMIUM_ACTIVE",
    "planName": "Alight Motion Pro / Member",
    "subscriptionType": "Yearly VIP License",
    "orderId": "GPA.ENCH-3974-4453-52315",
    "validUntil": "11 September 2027",
    "validUntilTimestamp": 1820622345678,
    "idToken": "eyJhbGciOiJSUzI1NiIs...",
    "refreshToken": "AMf-vBx..."
  }
}
Contoh cURL
curl -X POST https://ampro.enchant.id/api/verify-link \
  -H "Content-Type: application/json" \
  -d '{
    "email": "user@gmail.com",
    "magicLink": "LINK_DARI_EMAIL",

Contoh Integrasi di Server Luar

Script siap pakai untuk dipanggil dari bot WA / Telegram / backend kamu.

const axios = require('axios')
const BASE = 'https://ampro.enchant.id/api'

async function kirim(email) {
  const r = await axios.post(`${BASE}/send-link`, { email })
  return r.data
}

async function aktivasi(email, link, nickname) {
  const r = await axios.post(`${BASE}/verify-link`, {
    email,
    magicLink: link,
    nickname: nickname || undefined
  })
  return r.data
}
import requests

BASE = "https://ampro.enchant.id/api"

# 1. Kirim link
def send_link(email):
    return requests.post(f"{BASE}/send-link", json={"email": email}).json()

# 2. Verifikasi + Premium
def verify(email, link, nickname=None):
    payload = {"email": email, "magicLink": link}
    if nickname:
        payload["nickname"] = nickname
    return requests.post(f"{BASE}/verify-link", json=payload).json()
<?php
$base = "https://ampro.enchant.id/api";

function verifyLink($email, $link, $nickname = null) {
    global $base;
    $ch = curl_init("$base/verify-link");
    $data = json_encode([
        "email" => $email,
        "magicLink" => $link,
        "nickname" => $nickname
    ]);
    curl_setopt($ch, CURLOPT_POSTFIELDS, $data);
    curl_setopt($ch, CURLOPT_HTTPHEADER, ['Content-Type: application/json']);
    curl_setopt($ch, CURLOPT_RETURNTRANSFER, true);
    $res = curl_exec($ch);
    curl_close($ch);
    return json_decode($res, true);
}
